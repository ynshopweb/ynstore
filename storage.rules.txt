// ============================================================
// CONTOH FIREBASE STORAGE RULES untuk YN Shop
// Salin ke Firebase Console -> Storage -> Rules
// ============================================================
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    function isSignedIn() {
      return request.auth != null;
    }

    // QRIS pembayaran - siapa saja boleh lihat, hanya user login yang
    // terverifikasi admin (dicek lewat custom claim atau Firestore) boleh upload.
    // Catatan: Storage Rules tidak bisa langsung baca Firestore biasa,
    // jadi cara paling aman adalah set Custom Claim "admin": true lewat
    // Firebase Admin SDK saat akun admin dibuat, lalu cek di sini:
    match /payment/qris/{fileName} {
      allow read: if true;
      allow write: if isSignedIn() && request.auth.token.admin == true
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
    }

    // Bukti transfer customer - siapa saja yang login boleh upload punya sendiri
    match /proofs/{userId}/{fileName} {
      allow read: if true;
      allow write: if isSignedIn() && request.auth.uid == userId;
    }
  }
}
