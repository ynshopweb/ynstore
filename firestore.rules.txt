// ============================================================
// FIRESTORE SECURITY RULES untuk YN Shop (versi diperbaiki)
// Salin ke Firebase Console -> Firestore Database -> Rules -> Publish
//
// Ganti nilai APP_ID di bawah sesuai appId asli project Anda
// (lihat js/config.js -> appId). Default project ini: 'ynstore-default-id'
// ============================================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    // appId di-hardcode di sini karena /settings/{docId} tidak berada
    // di bawah path /artifacts/{appId}/, jadi variabel appId tidak bisa
    // diambil otomatis dari path saat function ini dipanggil dari sana.
    function isAdmin() {
      return isSignedIn() &&
        get(/databases/$(database)/documents/artifacts/ynstore-default-id/users/$(request.auth.uid)/profile/data).data.role == 'admin';
    }

    // Pengaturan pembayaran (QRIS) - hanya admin yang boleh menulis
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Produk - baca untuk semua, tulis hanya admin
    match /artifacts/{appId}/products/{productId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Profil user - hanya pemilik akun & admin yang boleh baca/tulis
    match /artifacts/{appId}/users/{userId}/profile/{docId} {
      allow read: if isSignedIn() && (request.auth.uid == userId || isAdmin());
      allow create: if isSignedIn() && request.auth.uid == userId;
      allow update: if isAdmin() || (isSignedIn() && request.auth.uid == userId);
    }

    // Pesanan - customer boleh buat pesanan sendiri & baca pesanan sendiri,
    // hanya admin yang boleh ubah status pesanan siapa pun
    match /artifacts/{appId}/orders/{orderId} {
      allow read: if true;
      allow create: if isSignedIn();
      allow update: if isAdmin();
    }
  }
}
